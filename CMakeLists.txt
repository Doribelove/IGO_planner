cmake_minimum_required(VERSION 3.10)
project(igo_local_planner)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp nav_core costmap_2d pluginlib tf2 tf2_ros tf2_geometry_msgs nav_msgs geometry_msgs std_msgs dynamic_reconfigure)
generate_dynamic_reconfigure_options(cfg/IGO.cfg)
catkin_package(INCLUDE_DIRS include LIBRARIES igo_local_planner CATKIN_DEPENDS roscpp nav_core costmap_2d pluginlib tf2 tf2_ros tf2_geometry_msgs nav_msgs geometry_msgs std_msgs dynamic_reconfigure)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(igo_local_planner src/core.cpp src/ros_planner.cpp)
add_dependencies(igo_local_planner ${PROJECT_NAME}_gencfg ${catkin_EXPORTED_TARGETS})
target_link_libraries(igo_local_planner ${catkin_LIBRARIES})
add_executable(igo_scenarios test/scenarios.cpp src/core.cpp)
add_executable(igo_plugin_load_check test/plugin_load_check.cpp)
add_dependencies(igo_plugin_load_check igo_local_planner ${catkin_EXPORTED_TARGETS})
target_link_libraries(igo_plugin_load_check ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  # These two master-free checks are CTest tests. Run from the workspace with:
  # (cd build && ctest -R '^igo_' --output-on-failure)
  add_test(NAME igo_scenarios COMMAND $<TARGET_FILE:igo_scenarios>)
  add_test(NAME igo_plugin_load_check COMMAND ${CATKIN_ENV} $<TARGET_FILE:igo_plugin_load_check>)
endif()
catkin_install_python(PROGRAMS scripts/ros_simulation.py scripts/run_simulation.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS igo_local_planner igo_scenarios igo_plugin_load_check ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(FILES igo_plugin.xml README.md LICENSE DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY config launch docs DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
